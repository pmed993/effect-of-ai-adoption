SELECT
    f.costat,
    f.curcdq,
    f.fyearq,
    f.datafmt,
    f.indfmt,
    f.consol,
    f.gvkey,
    f.datadate,
    f.tic,
    f.exchg,
    f.datacqtr,
    f.datafqtr,
    f.actq,
    f.atq,
    f.cheq,
    f.cshiq,
    f.cshopq,
    f.cshprq,
    f.dlcq,
    f.dlttq,
    f.dvpq,
    f.intanq,
    f.niq,
    f.xrdq,
    f.xsgaq,
    f.capxy,
    f.dvpy,
    f.dvy,
    f.oancfy,
    f.prstkcy,
    f.xrdy,
    f.cshtrq,
    f.dvpspq,
    f.dvpsxq,
    f.mkvaltq,
    f.prccq,
    f.tstkq,
    f.tstknq,
    f.prcraq,
    f.req,

    c.busdesc,
    c.county,
    c.naics,
    c.sic,
    c.spcindcd,

    a.emp,
    a.sale,
    a.oiadp, 
    a.capx,
    a.xrd, 
    a.at

FROM comp.fundq AS f

LEFT JOIN comp.company AS c
    ON f.gvkey = c.gvkey

LEFT JOIN comp.funda AS a
    ON f.gvkey = a.gvkey
   AND a.fyear = f.fyearq

WHERE f.indfmt = 'INDL'
  AND f.datafmt = 'STD'
  AND f.consol = 'C'
  AND f.costat = 'A'
  AND f.datadate BETWEEN '2005-01-01' AND '2027-02-01';
